motley_cue.conf¶

## Configuration for motley_cue

[mapper]
## various service-specific settings
##
## log_level -- default: WARNING
##
## DEBUG is verbose about the authorisation and assurance decisions, which is
## what you want when a user lands on the wrong shell tier. It does NOT print
## Access Tokens or OTPs -- those are reduced to a short fingerprint, and flaat's
## own token logging is clamped to INFO -- so it is safe to turn on temporarily
## on a running service.
##
## The decisions themselves (which shell tier, and why) are logged at AUDIT,
## which is above CRITICAL and therefore visible at every log_level. Look there
## first; you may not need DEBUG at all.
log_level = WARNING
## log_file -- default: /dev/stderr, ends up in syslog
# log_file = /var/log/motley_cue/mapper.log
##
## enable swagger documentation -- default: False
# enable_docs = False
## location of swagger docs -- default: /docs
# docs_url = /docs
## location of redoc docs -- default: /redoc
# redoc_url = /redoc
##
## API version -- default: v1
## supported versions: v1
# api_version = v1

[mapper.otp]
## use one-time passwords (OTP) instead of tokens as ssh password -- default: True
## this can be used when access tokens are too long to be used as passwords (>1k)
# use_otp = True
##
## backend for storing the OTP-AT mapping -- default: sqlite
## supported backends: memory, sqlite, sqlitedict
## memory: stores the mapping in memory, will be lost on restart; cannot be used in
##         a multi-process environment, e.g. with gunicorn
## sqlite: stores the mapping in a sqlite database; the only backend safe for
##         multi-process servers (atomic one-time pop shared by all workers)
## sqlitedict: stores the mapping in a sqlite database, but uses a python dict
##             as cache, so that the database is only accessed on changes
# backend = sqlite
##
## location for storing token database -- default: /var/lib/motley_cue/tokenmap.db
## only used when backend is sqlite or sqlitedict
## a prefix of ${backend}_ is added to the filename to avoid conflicts
## non-root processes fall back to ~/.config/motley_cue/tokenmap.db
# db_location = /var/lib/motley_cue/tokenmap.db
## path to file containing key for encrypting token db -- default: /var/lib/motley_cue/motley_cue.key
## key must be a URL-safe base64-encoded 32-byte key, and it will be created if it doesn't exist
## non-root processes fall back to ~/.config/motley_cue/motley_cue.key
# keyfile = /var/lib/motley_cue/motley_cue.key
##
## DO NOT put either of these in a world-writable directory such as /tmp. The
## database holds users' Access Tokens and the keyfile decrypts them, so anyone
## who can create the keyfile first decides the key, and anyone who can read it
## can impersonate every user of this service. motley_cue refuses to start if the
## keyfile is a symlink, is not owned by the service user, or is readable by
## group or other.


[privacy]
## configuration related to privacy policy
##
## contact information for service operator to be included in privacy policy -- default: NOT CONFIGURED
## this is an email address and MUST be filled in
# privacy_contact = NOT CONFIGURED
##
## privacy policy location (markdown file) -- default: /etc/motley_cue/privacystatement.md
# privacy_file = /etc/motley_cue/privacystatement.md


[DEFAULT]
## Magic section that provides default values for all sections below.
## These values are overwritten by values in each section, if specified.
## DO NOT CHANGE this section unless you really want to apply the changes to ALL OPs.
##
## this is the only key without a default value, must be specified for each section
# op_url =

## scopes required from the OP to access the service -- default: ["openid", "profile", "email", "eduperson_entitlement"]
## sane defaults for each OP are provided in the examples below
scopes = ["openid", "profile", "email", "eduperson_entitlement"]

## USER AUTHORISATION
## authorise all users from trusted OP, defaults to False if not specified
authorise_all = False

## list of VOs whose users are authorised to use the service
authorised_vos = []

## the OIDC claim containing the VOs specified above
vo_claim = eduperson_entitlement
# vo_claim = entitlements

## how many VOs need to be matched from the list, valid options: all, one, or an int
## defaults to one if not specified
vo_match = one

## list of individual users authorised to use the service
## specified through OIDC 'sub', relative to the section's OP ('iss')
## defaults to empy list if not specified
authorised_users = []

## audience claim specific to this service (OPTIONAL); it can be a string or a list of strings
## if empty or not specified, audience checking will not be used for authorisation
## !! be aware that many OPs do not support this feature, in which case it will be ignored for authorisation
##
## Worth understanding what leaving this empty means: a token is then accepted
## here no matter which service it was issued for. Every other relying party of
## the same OP that a user logs in to therefore ends up holding a credential
## that also works against this service -- and with authorise_all = True, that
## credential is a shell account on this host. Setting an audience is what binds
## a token to this service; motley_cue logs a warning at startup for any OP that
## has authorise_all set without one.
# audience = ssh_localhost

## You can find out your 'sub' and VOs you are a member of by using flaat:
##      $ pip install flaat
##      $ flaat-userinfo $TOKEN
## where $TOKEN contains an Access Token from the OP you are interested it.
## A commandline tool to retrieve access tokens is:
##      oidc-agent [https://github.com/indigo-dc/oidc-agent]

## ADMIN AUTHORISATION
##
## Each OP includes authorisation for the /admin endpoint,
## which allows suspending and resuming access for given users.
##
## Currently, only individual admins can be authorised (by 'sub' claim).
## This is meant for infrastructure security contacts.
## In the future, we plan to support this feature for VO managers too, to suspend
## members of their own VO (by using VO roles encoded in AARC-G002 entitlements).
## list of authorised admins
authorised_admins = []

## allow admins to suspend users from any OP
## defaults to False, which means an admin can only suspend users from their own OP
# authorise_admins_for_all_ops = False
##
##################################################################################
## ASSURANCE-BASED SHELL TIERS (OPT-IN -- disabled by default)
##
## motley_cue evaluates each user's assurance/MFA claims against the tier
## expressions below (highest privilege first: full, then limited, then
## restricted) and passes the resulting tier to feudalAdapter, which maps it to a
## login shell (see [backend.local_unix] shell / shell_limited / shell_restricted
## in feudal_adapter.conf). Assurance does not reject users: everyone who is
## authorised is deployed, the tier only decides the shell.
##
## Like every other option here, these can be set once in [DEFAULT] and
## overridden by any individual [authorisation.<op>] section -- which is the
## point: OPs differ in what assurance they can actually assert.
##
## By default (all tier expressions left unset) every user resolves to the "full"
## tier and gets feudalAdapter's default shell -- i.e. unchanged behaviour.
## Uncomment and adjust the lines below to enable the feature. Once an OP has a
## policy, a user matching none of its expressions gets the LEAST privileged
## tier rather than "full" (see assurance_based_shell_default_tier below), so a
## policy that never matches cannot silently hand out a full shell. Adding
## `assurance_based_shell_tier_restricted = *` as an explicit catch-all is still
## recommended, so the intent is visible in the config.
##
## An OP is only affected by a policy it actually inherits: options set here in
## [DEFAULT] apply to every OP (and to unknown ones), while options set in a
## single [authorisation.<op>] section leave all other OPs untouched.
##################################################################################

## assurance_prefix -- prefix for relative claim values
## default: https://refeds.org
# assurance_prefix = https://refeds.org

## assurance_claims -- token claims whose values are unioned into the evaluated
## assurance set. Each claim is searched in ALL available sources: the userinfo
## endpoint, the access token body (if it is a JWT) and token introspection.
## default: [assurance, eduperson_assurance, acr]
# assurance_claims = [assurance, eduperson_assurance, acr]

## Tier expressions. Grammar: E -> E "&" E | E "|" E | "(" E ")" | string, where
## '&' binds stronger than '|'. A string matches if the user's assurance set
## contains it verbatim OR prefixed with assurance_prefix -- so both full REFEDS
## URLs and bare values (e.g. an acr of '1') work. "+" = any claim present,
## "*" = always matches.
##
## CAREFUL: relative tokens are resolved against assurance_prefix, which is the
## refeds ROOT -- so `assurance/profile/cappuccino` and `profile/mfa` both work,
## but `profile/cappuccino` does NOT (the cappuccino profile lives under
## /assurance/). If you narrow assurance_prefix, a relative token that no longer
## resolves expands to a URL no OP asserts and silently never matches; spelling
## a value out in full (https://refeds.org/profile/mfa) always works whatever
## the prefix.

## users with MFA and the cappuccino profile get the full shell
# assurance_based_shell_tier_full = profile/mfa | assurance/profile/cappuccino

## users with the cappuccino profile but no MFA get a limited shell
# assurance_based_shell_tier_limited = assurance/IAP/medium

## everyone else (always matches) -- message-and-exit shell
# assurance_based_shell_tier_restricted = *

## assurance_based_shell_default_tier -- tier used if no expression matches.
## Left unset it is decided per OP: "full" for an OP with no tier expressions
## at all (feature off, unchanged behaviour), and the least privileged tier
## ("restricted") for an OP that does have a policy. Set it explicitly to
## override -- including back to "full", if you deliberately want unmatched
## users of an OP with a policy to keep the default shell.
## default: empty, i.e. decide automatically as described above
# assurance_based_shell_default_tier = restricted

## assurance_based_shell_max_tier -- caps the resulting tier. It can only ever
## lower a tier, never raise one. Most useful per-OP (see [authorisation.google]
## below); in [DEFAULT] it acts as a global "turn everything down" switch.
## default: empty, i.e. no cap
# assurance_based_shell_max_tier = limited



#########################################################################################
## AUTHORISATION PER OIDC PROVIDER
##
## The following sections are used to configure auhtorisation of users from multiple OPs.
## Any section named [authorisation.*] configures the authorisation for a single OP.
##
## Feel free to add more sections to support your desired OPs.
## You can overwrite any of the default values by simply specifying a new value.
## Take a look at the examples in the comments to help get you started.
#########################################################################################
