motley_cue.env¶

# Put the bundled virtualenv's bin first so any subprocess that calls a bare
# "python3"/"python" uses the venv interpreter (>= 3.10) instead of the distro
# default (e.g. 3.9 on EL9). The standard system paths are kept so helper tools
# (useradd, etc. used for account provisioning) stay reachable. Written out in
# full because systemd's EnvironmentFile does not expand $PATH.
PATH=/usr/lib/motley-cue/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
ERROR_LOG=/var/log/motley_cue/motley_cue.gunicorn.error
ACCESS_LOG=/var/log/motley_cue/motley_cue.gunicorn.access
BIND=unix:/run/motley_cue/motley-cue.sock
LOG_LEVEL_GUNICORN=info
FLAAT_CONFIG=/etc/motley_cue/flaat.conf
FEUDAL_ADAPTER_CONFIG=/etc/motley_cue/feudal_adapter.conf
MAX_REQUESTS=1000
MAX_REQUESTS_JITTER=50
